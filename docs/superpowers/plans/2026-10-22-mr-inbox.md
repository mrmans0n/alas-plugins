# Nacho's MR Inbox Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A new Alas plugin `plugins/nacho-mr-inbox`: a tab and a sidebar listing the current GitLab project's open merge requests by what they need, its merge train, your own MRs, add-to-train / retry actions, and CI Doctor on GitLab CI.

**Architecture:** Copy `plugins/nacho-pr-inbox` and rewrite each module for GitLab, keeping file names, module boundaries and (inside CI Doctor) type names, so each module's diff stays reviewable against its pr-inbox original. All GitLab access goes through `glab` processes declared in `plugin.json`. `glab` resolves the host, login and project from the repository's own remotes (`:fullpath` / `:id` placeholders), so no host is ever named in code.

**Tech Stack:** TypeScript (type-stripped by Node ≥ 22.18 for tests, bundled by esbuild), `@alas/plugin` SDK API 15, `node --test`, `glab` ≥ 1.40 on the execution host.

**Spec:** `docs/superpowers/specs/2026-10-22-mr-inbox-design.md`

## Global Constraints

- Never write a real self-managed GitLab host, group, project, user or branch name in code, tests, fixtures, docs or commits. Use `gitlab.com` or `gitlab.example.com`, group `acme`, project `acme/shop`, users `nacho`, `ana`, `bob`.
- Plugin id `io.nlopez.nacho-mr-inbox`, name "Nacho's MR Inbox", version `0.1.0`, `"api": 15`, `"remote": true`.
- `network`: exactly `["gitlab.com", "secure.gravatar.com"]`. Avatars from any other host fall back to initials.
- `nacho-pr-inbox` is not modified.
- Plugin runtime has no `URL`, `TextEncoder`, `console` or timers: validate strings with anchored regexes, as pr-inbox does.
- Every appended process argument is built by plugin code and matched against an anchored pattern before `processRun`.
- Limits copied from the spec: 50 open MRs, 50 merged candidates, 14-day CI window, ≤ 30 pipelines per query in pages of 10, ≤ 5 comparison pipelines per side, ≤ 100 jobs per pipeline in pages of 25, ≤ 4 comparison logs, ≤ 8 cache entries in 256 KiB, brief < 32 KiB, trace tail 262144 bytes, at most 2 processes at once.
- Commits: conventional style `feat(nacho-mr-inbox): …`, ending with `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.
- Run commands from the repository root (`alas-plugins/`) unless a step says otherwise.
- Until Task 7, `npm test` for the whole plugin fails: un-migrated GitHub tests import changed modules. Each task runs only its own test files. Task 7 deletes nothing; it is the first task where `npm test -w nacho-mr-inbox` must pass in full.

```ts
import type { Outcome, ProcessResult } from "@alas/plugin";

export interface Progress { done: number; running: number; total: number }
export interface Pipeline { id: number; status: string; sha: string; progress: Progress | null }

export interface MergeRequest {
  iid: number;
  title: string;
  url: string;
  draft: boolean;
  author: string;
  /** Only from the manifest's `network` hosts; others fall back to initials. */
  avatar?: string;
  branch: string;
  target: string;
  /** `diffHeadSha`. */
  sha: string;
  /** ISO 8601, so it sorts as text. */
  updatedAt: string;
  /** GitLab's `detailedMergeStatus`, such as `MERGEABLE`, `NOT_APPROVED` or `CI_STILL_RUNNING`. */
  status: string;
  conflicts: boolean;
  approvalsLeft: number;
  approvalsRequired: number;
  /** Reviewers whose review state is `REQUESTED_CHANGES`. */
  changesRequestedBy: string[];
  /** Resolvable threads not yet resolved. */
  unresolved: number;
  pipeline: Pipeline | null;
}

export interface Car { iid: number; status: string; pipeline: { status: string; progress: Progress | null } | null }
export interface Train { target: string; cars: Car[] }

export interface Merged { iid: number; title: string; url: string; author: string; avatar?: string; branch: string; mergedAt: string }

export interface Inbox {
  /** `group/sub/project`. */
  project: string;
  webUrl: string;
  /** From `webUrl`: every link must be on it. */
  host: string;
  /** The `glab` user. "Yours" means authored by them. */
  user: string;
  mrs: MergeRequest[];
  /** Non-empty trains, cars in merge order. */
  trains: Train[];
  /** The most recently merged first. */
  merged: Merged[];
}

export const MERGED_RECENT_MS = 24 * 3600_000;
export const MERGED_MIN = 5;

export function recentMerged(merged: Merged[], now: number): Merged[] {
  const older = merged.findIndex((m) => !(now - Date.parse(m.mergedAt) <= MERGED_RECENT_MS));
  return merged.slice(0, Math.max(MERGED_MIN, older === -1 ? merged.length : older));
}

export type Bucket = "ready" | "blocked" | "failing" | "drafts";
export const BUCKETS: Bucket[] = ["ready", "blocked", "failing", "drafts"];

export const SIGN_IN = "Sign in with `glab auth login --hostname <host>` in a terminal, then Refresh.";
export const NO_REMOTE = "This repository has no GitLab remote glab knows. For self-managed GitLab, run `glab auth login --hostname <host>` first.";
export const NO_GLAB = "glab is not installed on this host. Install it, sign in with `glab auth login`, then Refresh.";

const isObject = (v: unknown): v is Record<string, any> => typeof v === "object" && v !== null && !Array.isArray(v);
const str = (v: unknown): string | null => (typeof v === "string" ? v : null);
const count = (v: any): number => (Number.isInteger(v?.count) && v.count > 0 ? v.count : 0);
const nat = (v: unknown): number => (Number.isInteger(v) && (v as number) > 0 ? (v as number) : 0);
/** GraphQL iids are strings. */
const iidOf = (v: unknown): number | undefined => (typeof v === "string" && /^[1-9]\d{0,9}$/.test(v) ? Number(v) : undefined);

/** `gid://gitlab/Ci::Pipeline/235387` → 235387. */
export function pipelineId(gid: unknown): number | undefined {
  const m = typeof gid === "string" ? /^gid:\/\/gitlab\/Ci::Pipeline\/([1-9]\d{0,15})$/.exec(gid) : null;
  return m ? Number(m[1]) : undefined;
}

// No `URL` in the plugin runtime: strict patterns, so no userinfo, port or other host slips in.
const AVATAR = /^https:\/\/(?:secure\.gravatar\.com\/avatar\/|gitlab\.com\/uploads\/)[A-Za-z0-9/?=&%._-]*$/;
function avatar(author: any): { avatar?: string } {
  const url = str(author?.avatarUrl);
  return url !== null && url.length <= 512 && AVATAR.test(url) ? { avatar: url } : {};
}

function progress(p: any): Progress | null {
  const total = count(p?.jobs);
  if (total === 0) return null;
  const running = Math.min(total, count(p.running));
  return { done: Math.min(total - running, count(p.done)), running, total };
}

function mergeRequest(node: unknown, base: string): MergeRequest | undefined {
  if (!isObject(node)) return undefined;
  const iid = iidOf(node.iid);
  const { title, webUrl, draft, updatedAt, sourceBranch, targetBranch } = node;
  if (iid === undefined || typeof title !== "string" || typeof webUrl !== "string" || !webUrl.startsWith(base)) return undefined;
  if (typeof draft !== "boolean" || typeof updatedAt !== "string" || typeof sourceBranch !== "string" || typeof targetBranch !== "string") return undefined;
  const p = node.headPipeline;
  const id = pipelineId(p?.id);
  const reviewers = Array.isArray(node.reviewers?.nodes) ? node.reviewers.nodes : [];
  return {
    iid, title, url: webUrl, draft,
    author: str(node.author?.username) ?? "ghost", ...avatar(node.author),
    branch: sourceBranch, target: targetBranch, sha: str(node.diffHeadSha) ?? "", updatedAt,
    status: str(node.detailedMergeStatus) ?? "UNCHECKED",
    conflicts: node.conflicts === true,
    approvalsLeft: nat(node.approvalsLeft), approvalsRequired: nat(node.approvalsRequired),
    changesRequestedBy: reviewers.filter((r: any) => r?.mergeRequestInteraction?.reviewState === "REQUESTED_CHANGES" && typeof r.username === "string").map((r: any) => r.username),
    unresolved: Math.max(0, nat(node.resolvableDiscussionsCount) - nat(node.resolvedDiscussionsCount)),
    pipeline: id !== undefined && typeof p.status === "string" ? { id, status: p.status, sha: str(p.sha) ?? "", progress: progress(p) } : null,
  };
}

function trains(nodes: unknown): Train[] {
  if (!Array.isArray(nodes)) return [];
  return nodes.flatMap((t: any) => {
    if (typeof t?.targetBranch !== "string" || !Array.isArray(t.cars?.nodes)) return [];
    const cars: Car[] = t.cars.nodes.flatMap((c: any) => {
      const iid = iidOf(c?.mergeRequest?.iid);
      if (iid === undefined || typeof c.status !== "string") return [];
      return [{ iid, status: c.status, pipeline: typeof c.pipeline?.status === "string" ? { status: c.pipeline.status, progress: progress(c.pipeline) } : null }];
    });
    return cars.length ? [{ target: t.targetBranch, cars }] : [];
  });
}

function merged(nodes: unknown, base: string): Merged[] {
  if (!Array.isArray(nodes)) return [];
  const out = new Map<number, Merged>();
  for (const n of nodes) {
    const iid = iidOf(n?.iid);
    if (iid === undefined || out.has(iid) || typeof n.title !== "string" || typeof n.webUrl !== "string" || !n.webUrl.startsWith(base) || typeof n.sourceBranch !== "string" || typeof n.mergedAt !== "string") continue;
    out.set(iid, { iid, title: n.title, url: n.webUrl, author: str(n.author?.username) ?? "ghost", ...avatar(n.author), branch: n.sourceBranch, mergedAt: n.mergedAt });
  }
  return [...out.values()];
}

/** The list reply, or the message to show when it is not one. */
export function parseInbox(stdout: string): Inbox | string {
  let json: any;
  try { json = JSON.parse(stdout); } catch { return "glab returned an unexpected reply."; }
  const firstError = Array.isArray(json?.errors) ? str(json.errors[0]?.message) : null;
  if (firstError) return `GitLab: ${firstError.slice(0, 300)}`;
  const project = json?.data?.project;
  if (!isObject(project)) return "GitLab returned no project. Check that glab can read it.";
  const host = /^https:\/\/([a-z0-9.-]+(?::\d{1,5})?)\/[^\s]+$/i.exec(str(project.webUrl) ?? "")?.[1]?.toLowerCase();
  if (!host || typeof project.fullPath !== "string" || !Array.isArray(project.mergeRequests?.nodes)) return "glab returned an unexpected reply.";
  const base = `https://${host}/`;
  const mrs = new Map<number, MergeRequest>();
  for (const n of project.mergeRequests.nodes) {
    const mr = mergeRequest(n, base);
    if (mr && !mrs.has(mr.iid)) mrs.set(mr.iid, mr);
  }
  return {
    project: project.fullPath, webUrl: project.webUrl, host, user: str(json.data.currentUser?.username) ?? "",
    mrs: [...mrs.values()], trains: trains(project.mergeTrains?.nodes), merged: merged(project.merged?.nodes, base),
  };
}

/** `glab api projects/:id`'s `merge_trains_enabled`; `undefined` for an unreadable reply. */
export function parseTrainsEnabled(stdout: string): boolean | undefined {
  try {
    const json = JSON.parse(stdout);
    return isObject(json) ? json.merge_trains_enabled === true : undefined;
  } catch { return undefined; }
}

export function inTrain(inbox: Inbox, mr: MergeRequest): boolean {
  return inbox.trains.some((t) => t.target === mr.target && t.cars.some((c) => c.iid === mr.iid));
}

export const isFailing = (mr: MergeRequest) => mr.pipeline?.status === "FAILED" || mr.pipeline?.status === "CANCELED";

/** GitLab already decided: mergeable now, or mergeable once the running pipeline passes. */
export function addMode(mr: MergeRequest): "now" | "whenGreen" | undefined {
  if (mr.draft || isFailing(mr)) return undefined;
  if (mr.status === "MERGEABLE") return "now";
  if (mr.status === "CI_STILL_RUNNING") return "whenGreen";
  return undefined;
}

export function bucketOf(mr: MergeRequest): Bucket {
  if (mr.draft) return "drafts";
  if (isFailing(mr)) return "failing";
  if (mr.status === "MERGEABLE") return "ready";
  return "blocked";
}

/** MRs by bucket, newest-updated first, without the ones already in a train. */
export function classify(inbox: Inbox): Record<Bucket, MergeRequest[]> {
  const out: Record<Bucket, MergeRequest[]> = { ready: [], blocked: [], failing: [], drafts: [] };
  const sorted = inbox.mrs.filter((m) => !inTrain(inbox, m)).sort((a, b) => (a.updatedAt < b.updatedAt ? 1 : a.updatedAt > b.updatedAt ? -1 : 0));
  for (const m of sorted) out[bucketOf(m)].push(m);
  return out;
}

export function updatedLabel(fetchedAt: number | undefined, now: number): string | undefined {
  if (fetchedAt === undefined) return undefined;
  const seconds = (now - fetchedAt) / 1000;
  if (seconds < 60) return "Updated just now";
  if (seconds < 3600) return `Updated ${Math.floor(seconds / 60)}m ago`;
  return `Updated ${Math.floor(seconds / 3600)}h ago`;
}

export function agoLabel(iso: string, now: number): string | undefined {
  const at = Date.parse(iso);
  if (Number.isNaN(at)) return undefined;
  const seconds = Math.max(0, (now - at) / 1000);
  if (seconds < 60) return "just now";
  if (seconds < 3600) return `${Math.floor(seconds / 60)}m ago`;
  if (seconds < 86400) return `${Math.floor(seconds / 3600)}h ago`;
  return `${Math.floor(seconds / 86400)}d ago`;
}

/** glab boxes errors in an `ERROR` banner: skip it and blank lines. */
function firstLine(text: string): string | undefined {
  const line = text.split("\n").map((l) => l.trim()).find((l) => l !== "" && l !== "ERROR");
  return line === undefined ? undefined : line.slice(0, 300);
}

export function processError(outcome: Outcome<ProcessResult>): string | undefined {
  if (outcome.error) return /command not found|No such file/i.test(outcome.error.message) ? NO_GLAB : outcome.error.message;
  const { exit, stderr, stdout, timedOut } = outcome.result;
  if (timedOut) return "glab timed out.";
  if (exit === 0) return undefined;
  if (/glab: (?:command )?not found/.test(stderr)) return NO_GLAB;
  if (/none of the git remotes configured|not a git repository/.test(stderr)) return NO_REMOTE;
  if (/401 Unauthorized|glab auth login|no token found/i.test(stderr)) return SIGN_IN;
  return firstLine(stderr) ?? firstLine(stdout) ?? `glab exited with ${exit}.`;
}

import { addMode, BUCKETS, classify, inTrain, isFailing, recentMerged, type Bucket, type Car, type Inbox, type MergeRequest, type Progress } from "./inbox.ts";

export interface ViewState {
  inbox?: Inbox;
  fetchedAt?: number;
  refreshing: boolean;
  error?: string;
  /** From the project reply; `undefined` until known, which counts as on. */
  trainsEnabled?: boolean;
  /** The MR being added or merged. */
  busy?: number;
  /** The pipeline being retried. */
  retrying?: number;
  /** The last write failure per MR iid. */
  errors: Record<number, string>;
  addAll?: { total: number; done: number };
}

export type Tone = "accent" | "success" | "warn" | "danger" | "dim";
export interface Chip { text: string; tone: Tone }
export interface AddAction { when: "now" | "whenGreen"; look: "go" | "plain"; label: string; short: string; disabled: boolean; busy: boolean }

export interface Row {
  iid: number; title: string; url: string; branch: string; author: string; avatar?: string; updatedAt: string;
  mine: boolean;
  chips: Chip[];
  /** Pipeline still running: how far along. */
  progress?: Progress & { failed: boolean };
  add?: AddAction;
  retry?: { pipeline: number; busy: boolean; disabled: boolean };
  /** There is a head pipeline to inspect. */
  doctor: boolean;
  error?: string;
}

export interface Section { key: Bucket; title: string; tone: Tone; rows: Row[] }
export interface CarView { iid: number; title: string; url?: string; mine: boolean; label: string; tone: Tone; progress?: Progress }
export interface TrainView { target: string; cars: CarView[] }
export interface MergedRow { iid: number; title: string; url: string; branch: string; author: string; avatar?: string; mergedAt: string }

export interface InboxModel {
  project?: string;
  fetchedAt?: number;
  refreshing: boolean;
  error?: string;
  loading: boolean;
  trainsEnabled: boolean;
  trains: TrainView[];
  counts: Record<Bucket, number>;
  sections: Section[];
  /** The user's MRs outside the train, newest first: the rail's list. */
  mine: Row[];
  merged: MergedRow[];
  addAll?: { count: number; label: string; disabled: boolean; progress?: string };
}

const TITLES: Record<Bucket, string> = { ready: "Ready", blocked: "Blocked", failing: "Failing", drafts: "Drafts" };
const TONES: Record<Bucket, Tone> = { ready: "success", blocked: "warn", failing: "danger", drafts: "dim" };

/** Statuses a dedicated chip already explains, or that need no chip. */
const EXPLAINED = new Set(["MERGEABLE", "CI_STILL_RUNNING", "CI_MUST_PASS", "NOT_APPROVED", "DISCUSSIONS_NOT_RESOLVED", "CONFLICT", "DRAFT_STATUS", "REQUESTED_CHANGES", "NOT_OPEN"]);
const TRANSIENT = new Set(["CHECKING", "UNCHECKED", "PREPARING", "APPROVALS_SYNCING"]);
const NAMED: Record<string, string> = { NEED_REBASE: "needs rebase", BLOCKED_STATUS: "blocked by dependency", MERGE_REQUEST_BLOCKED: "blocked by dependency" };

export function chips(mr: MergeRequest): Chip[] {
  const out: Chip[] = [];
  const p = mr.pipeline;
  if (p?.status === "SUCCESS") out.push({ text: "✓ pipeline", tone: "success" });
  else if (p?.status === "FAILED") out.push({ text: "✗ pipeline", tone: "danger" });
  else if (p?.status === "CANCELED") out.push({ text: "pipeline canceled", tone: "danger" });
  else if (p?.status === "MANUAL") out.push({ text: "pipeline manual", tone: "dim" });
  else if (!p && mr.status === "CI_MUST_PASS") out.push({ text: "no pipeline", tone: "warn" });
  if (mr.changesRequestedBy.length) out.push({ text: `changes requested · ${mr.changesRequestedBy.map((u) => `@${u}`).join(", ")}`, tone: "danger" });
  else if (mr.status === "REQUESTED_CHANGES") out.push({ text: "changes requested", tone: "danger" });
  if (mr.unresolved > 0) out.push({ text: `💬 ${mr.unresolved} unresolved`, tone: "warn" });
  if (mr.approvalsLeft > 0) out.push({ text: `needs ${mr.approvalsLeft} approval${mr.approvalsLeft === 1 ? "" : "s"}`, tone: "warn" });
  else if (mr.approvalsRequired > 0) out.push({ text: "✓ approved", tone: "success" });
  if (mr.conflicts) out.push({ text: "conflicts", tone: "warn" });
  if (TRANSIENT.has(mr.status)) out.push({ text: "checking…", tone: "dim" });
  else if (NAMED[mr.status]) out.push({ text: NAMED[mr.status], tone: "warn" });
  else if (!EXPLAINED.has(mr.status)) out.push({ text: mr.status.toLowerCase().replace(/_/g, " "), tone: "warn" });
  return out;
}

function addAction(mr: MergeRequest, state: ViewState): AddAction | undefined {
  const when = addMode(mr);
  if (!when) return undefined;
  const trains = state.trainsEnabled !== false;
  const label = !trains ? "Merge when green" : when === "now" ? "🚂 Add to train" : "🚂 Add when green";
  const short = !trains ? "Merge" : when === "now" ? "Add" : "When green";
  return { when, look: when === "now" ? "go" : "plain", label, short, disabled: state.busy !== undefined, busy: state.busy === mr.iid };
}

function row(mr: MergeRequest, state: ViewState): Row {
  const p = mr.pipeline;
  const add = addAction(mr, state);
  const running = p?.progress && p.progress.done < p.progress.total && !isFailing(mr);
  return {
    iid: mr.iid, title: mr.title, url: mr.url, branch: mr.branch, author: mr.author, ...(mr.avatar ? { avatar: mr.avatar } : {}), updatedAt: mr.updatedAt,
    mine: mr.author === state.inbox?.user,
    chips: chips(mr),
    ...(running ? { progress: { ...p!.progress!, failed: false } } : {}),
    ...(add ? { add } : {}),
    ...(p && isFailing(mr) ? { retry: { pipeline: p.id, busy: state.retrying === p.id, disabled: state.retrying !== undefined } } : {}),
    doctor: p !== null,
    ...(state.errors[mr.iid] === undefined ? {} : { error: state.errors[mr.iid] }),
  };
}

function car(c: Car, inbox: Inbox): CarView {
  const mr = inbox.mrs.find((m) => m.iid === c.iid);
  const status = c.pipeline?.status;
  const [label, tone]: [string, Tone] =
    status === "FAILED" || status === "CANCELED" ? ["failed", "danger"]
    : c.status === "MERGING" || c.status === "MERGED" ? ["merging ✓", "success"]
    : c.status === "STALE" ? ["restarting", "dim"]
    : (status === "RUNNING" || status === "PENDING") && c.pipeline?.progress ? [`⏳ ${c.pipeline.progress.done}/${c.pipeline.progress.total}`, "warn"]
    : status === "RUNNING" || status === "PENDING" ? ["⏳ running", "warn"]
    : ["queued", "dim"];
  return { iid: c.iid, title: mr?.title ?? `!${c.iid}`, ...(mr ? { url: mr.url } : {}), mine: mr?.author === inbox.user, label, tone, ...(c.pipeline?.progress ? { progress: c.pipeline.progress } : {}) };
}

    } } : {}),

export class GitLab {
  blockedUntil = 0;
  authBlocked = false;
  private queue: ProcessQueue;
  readonly worktree: string;
  readonly project: string;
  readonly host: string;
  private now: () => number;
  private backoff: (seconds: number) => void;
  constructor(queue: ProcessQueue, worktree: string, project: string, host: string, now: () => number, backoff: (seconds: number) => void) {
    this.queue = queue; this.worktree = worktree; this.project = project; this.host = host; this.now = now; this.backoff = backoff;
    this.queue.setGate(() => !this.authBlocked && this.now() >= this.blockedUntil);
  }
  retry(): void { this.authBlocked = false; this.queue.resume(); }
  observe(outcome: Outcome<ProcessResult>): void {
    if (!processError(outcome)) return;
    const parsed = response(outcome.result?.stdout ?? "");
    const message = `${outcome.result?.stderr ?? ""}\n${parsed.body}\n${processError(outcome)}`;
    if (/\b429\b|rate limit|too many requests/i.test(message) || /^HTTP\/\S+ 429/m.test(parsed.headers)) {
      const retry = /retry-after:\s*(\d+)/i.exec(parsed.headers);
      const reset = /ratelimit-reset:\s*(\d+)/i.exec(parsed.headers);
      const delay = retry ? Number(retry[1]) * 1000 : reset ? Math.max(60000, Number(reset[1]) * 1000 - this.now()) : 60000;
      this.blockedUntil = Math.max(this.blockedUntil, this.now() + Math.max(1000, delay));
      this.backoff(Math.max(1, Math.ceil((this.blockedUntil - this.now()) / 1000)));
    }
    if (/401 Unauthorized|glab auth login|no token found/i.test(message)) this.authBlocked = true;
  }
  private gateError(): string | undefined {
    if (this.authBlocked) return "Sign in with glab auth login on this execution host, then Retry.";
    if (this.now() < this.blockedUntil) return `GitLab rate limited. Retry after ${new Date(this.blockedUntil).toISOString()}.`;
    return undefined;
  }
  private ready = () => !this.authBlocked && this.now() >= this.blockedUntil;
  json(rest: string, done: (r: Result<any>) => void, options: RequestOptions = {}): void {
    const blocked = this.gateError();
    if (blocked) return done({ error: blocked, complete: false });
    this.queue.run("ci-api", this.worktree, [endpoint(this.project, rest)], outcome => {
      this.observe(outcome);
      const error = processError(outcome);
      if (error) return done({ error, complete: false });
      if (outcome.result?.truncated) return done({ error: "GitLab JSON output was truncated; evidence is incomplete.", complete: false });
      let value: unknown;
      try { value = JSON.parse(response(outcome.result!.stdout).body); }
      catch { return done({ error: "GitLab returned malformed JSON; evidence is unavailable.", complete: false }); }
      done({ value, complete: true });
    }, { ...options, ready: this.ready });
  }

    this.json(`merge_requests/${iid}`, r => this.normalize(r, v => parseIdentity(v, this.project), done), options);

  run(id: number, done: (r: Result<Run>) => void, options: RequestOptions = {}): void {
    this.json(`pipelines/${id}`, r => this.normalize(r, v => parsePipeline(v, this.host, this.project), done), options);

  runs(filter: { mr?: number; ref?: string; since: string }, done: (r: Result<Run[]>) => void, options: RequestOptions = {}): void {

      const rest = filter.mr !== undefined

        : `pipelines?ref=${encodeURIComponent(filter.ref ?? "")}&source=push&updated_after=${encodeURIComponent(filter.since)}&per_page=10&page=${page}`;
      this.json(rest, r => {

        let values: Run[];
        try { values = parsePipelines(r.value, this.host, this.project); } catch (e) { return done({ value: collected, error: String(e), complete: false }); }
        collected.push(...values.filter(v => v.created >= filter.since));
        const more = values.length === 10;
        if (more && page < 3) return next(page + 1);
        done({ value: collected, complete: !more, warning: more ? "Pipeline history incomplete: stopped at 30 summaries." : undefined });
      }, options);
    };
    next(1);
  }
  jobs(pipeline: number, done: (r: Result<Job[]>) => void, options: RequestOptions = {}): void {
    const collected: Job[] = [];
    const next = (page: number) => this.json(`pipelines/${pipeline}/jobs?include_retried=true&per_page=25&page=${page}`, r => {
      if (r.error) return done({ value: collected, error: r.error, complete: false });
      let jobs: Job[];
      try { jobs = parseJobs(r.value, pipeline, this.host); } catch (e) { return done({ value: collected, error: String(e), complete: false }); }
      collected.push(...jobs);
      const more = jobs.length === 25;
      if (more && page < 4) return next(page + 1);
      // Attempt numbers count retries across pages, so recount once everything is in.
      for (const j of collected) j.attempt = collected.filter(o => o.name === j.name && o.id < j.id).length + 1;
      done({ value: collected, complete: !more, warning: more ? "Jobs incomplete: stopped at 100 jobs." : undefined });
    }, options);
    next(1);
  }

    this.queue.run("ci-trace", this.worktree, [String(job)], outcome => {
      this.observe(outcome);

      const out = outcome.result!;
      const headCut = out.truncated || out.stdout.length >= 262144;
      done({ value: { ...excerpt(out.stdout, headCut), fetchedAt: this.now() }, complete: !out.truncated });
    }, { ...options, ready: this.ready });
  }
  retryJob(id: number, done: (error?: string) => void): void { this.write(`jobs/${id}/retry`, done); }
  retryPipeline(id: number, done: (error?: string) => void): void { this.write(`pipelines/${id}/retry`, done); }

    this.queue.run("retry", this.worktree, [endpoint(this.project, rest)], outcome => { this.observe(outcome); done(processError(outcome)); }, { priority: true });
  }
  private normalize<T>(r: Result<any>, parse: (v: unknown) => T, done: (r: Result<T>) => void): void {
    if (r.error) return done({ error: r.error, complete: false });
    let value: T;
    try { value = parse(r.value); }
    catch (e) { return done({ error: String(e), complete: false }); }
    done({ value, complete: r.complete });
  }
}

  const body = el(c.url?.startsWith("https://") ? "a" : "span", `car ${tone(c.tone)}${c.mine ? " mine" : ""}`, el("span", "num", `!${c.iid}`), c.mine && el("span", "you", "YOU"), el("span", "car-label", c.label));
  if (body instanceof HTMLAnchorElement) { body.href = c.url!; body.rel = "noreferrer"; }
  body.title = title;
  return body;
}

function trainStrip(t: TrainView): HTMLElement {
  if (compact()) {
    return el("section", "section", el("h4", "section-head tone-accent", el("span", "dot"), `🚂 Train → ${t.target}`, el("span", "count", String(t.cars.length))),
      ...t.cars.map((c, i) => el("div", `stop ${tone(c.tone)}${c.mine ? " mine" : ""}`, el("span", "num", String(i + 1)), c.url ? link(`!${c.iid}`, c.url, "stop-link", c.title) : el("span", "", `!${c.iid}`), c.mine && el("span", "you", "YOU"), el("span", "grow"), el("span", "when", c.label))));
  }
  const cars = el("div", "train-cars");
  t.cars.forEach((c, i) => { if (i > 0) cars.append(el("span", "coupler")); cars.append(carEl(c, i)); });
  return el("section", "train", el("span", "loco", "🚂"), el("span", "branch", t.target), cars);
}
```

  `carEl` builds the anchor itself because `link()` only takes a text label; it follows `link()`'s rule that only `https` URLs become links.
